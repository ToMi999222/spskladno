Cvičení 02.3:

x = int(input("Zadejte první číslo: "  ))
y = input("Zadejte operaci: "  )
z = int(input("Zadejte druhé číslo: "  ))

if y == "deleni":
    print(x / z)
    if z == 0:
        print("Nelze dělit nulou")

elif y == "nasobeni":
    print(x * z)

elif y == "odcitani":
    print(x - z)

elif y == "scitani":
    print(x + z)
